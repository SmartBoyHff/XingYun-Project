# VR团队代码规范

## 一、命名规范

### 1.1 文件命名

| 类型 | 格式 | 示例 |
|------|------|------|
| 脚本文件 | `VR{模块}_{功能}.cs` | `VR4_TireGaugeController.cs` |
| 预制体 | `PVR{模块}_{名称}.prefab` | `PVR4_WorkLight.prefab` |
| 场景 | `Module{编号}_{功能}_VR.unity` | `Module4_Maintenance_VR.unity` |
| UI面板 | `UIVR_{功能}.prefab` | `UIVR_WorkOrder.prefab` |
| 数据资产 | `Data_VR_{类型}.asset` | `Data_VR_FaultList.asset` |

### 1.2 代码命名

| 类型 | 格式 | 示例 |
|------|------|------|
| 类名 | PascalCase | `VR4_TireGaugeController` |
| 接口 | `I`前缀 + PascalCase | `IVR_ToolInteractable` |
| 方法名 | PascalCase | `OnPressurizeButtonPressed()` |
| 公共变量 | PascalCase | `public float StandardPressure` |
| 私有变量 | camelCase | `private float currentPressure` |
| 序列化字段 | `_`前缀 + camelCase | `[SerializeField] private float _maxPressure` |
| 常量 | UPPER_SNAKE_CASE | `private const float GRAVITY = 9.8f;` |

### 1.3 模块编号

| 模块 | 编号 | 脚本前缀 |
|------|------|----------|
| 通用（模块0） | 00 | `VR00_` |
| 维护保养（模块4） | 04 | `VR4_` |
| 智能传感器（模块5） | 05 | `VR5_` |
| 整车制造（模块6） | 06 | `VR6_` |

## 二、脚本结构规范（重点：桌面小组复用友好）

### 2.1 标准脚本模板
// ============================================================
// 文件名：VR4_TireGaugeController.cs
// 模块：模块4 - 维护保养
// 功能：胎压表控制器，模拟胎压测量和加压过程
// 作者：[姓名]
// 创建日期：2024-XX-XX
// 最后更新：2024-XX-XX
// 
// 【桌面小组适配说明】
// 本脚本为VR版本实现，使用XR Interaction Toolkit。
// 桌面版本适配时，需要修改以下部分：
// 1. 输入方式：将XR手柄输入改为鼠标/触控笔输入
// 2. UI显示：World Space Canvas改为Screen Space Canvas
// 3. 高亮效果：Outline Shader可保留，交互触发方式改为鼠标悬停/点击
// ============================================================

using UnityEngine;
using UnityEngine.XR.Interaction.Toolkit;
using VRHelmet.Shared.Utils;
using VRHelmet.Shared.Events;

namespace VRHelmet.VRTeam.Maintenance
{
    /// <summary>
    /// 胎压表控制器
    /// 
    /// 【功能说明】
    /// 1. 连接轮胎气门嘴后显示当前胎压
    /// 2. 长按加压按钮连续加压，释放停止
    /// 3. 达到标准胎压时触发完成事件
    /// 4. 超过最大胎压时自动停止并警告
    /// 
    /// 【依赖组件】
    /// - XRBaseInteractor：用于射线交互
    /// - TextMeshProUGUI：显示胎压数值
    /// - AudioSource：播放加压音效
    /// 
    /// 【桌面小组适配指南】
    /// - 替换 XRBaseInteractor 为鼠标射线检测（Physics.Raycast）
    /// - 替换 XR Grab Interactable 为 IDragHandler 接口
    /// - 按钮按下/释放事件改为鼠标的 OnPointerDown/OnPointerUp
    /// - 其他业务逻辑完全复用，无需修改
    /// </summary>
    public class VR4_TireGaugeController : MonoBehaviour
    {
        #region 序列化字段 - 在Inspector中配置
        
        [Header("=== 参数设置 ===")]
        [Tooltip("标准胎压值（bar），达到此值触发完成事件")]
        [SerializeField] private float _standardPressure = 2.5f;
        
        [Tooltip("最大胎压值（bar），超过此值自动停止加压")]
        [SerializeField] private float _maxPressure = 3.0f;
        
        [Tooltip("初始胎压值（bar）")]
        [SerializeField] private float _initialPressure = 1.5f;
        
        [Tooltip("加压速度（bar/秒）")]
        [SerializeField] private float _pressurizeRate = 1.0f;
        
        [Header("=== 组件引用 ===")]
        [Tooltip("胎压表屏幕显示组件")]
        [SerializeField] private TextMeshProUGUI _displayText;
        
        [Tooltip("加压按钮（XR交互组件）")]
        [SerializeField] private XRSimpleInteractable _pressurizeButton;
        
        [Tooltip("连接气门嘴的触发器区域")]
        [SerializeField] private Collider _connectTrigger;
        
        [Header("=== 音效 ===")]
        [Tooltip("加压时的音效")]
        [SerializeField] private AudioClip _pressurizeSound;
        
        [Tooltip("达到标准胎压的音效")]
        [SerializeField] private AudioClip _completeSound;
        
        [Tooltip("超过最大胎压的警告音效")]
        [SerializeField] private AudioClip _warningSound;
        
        #endregion
        
        #region 私有变量
        
        private float _currentPressure;           // 当前胎压值
        private bool _isConnected;                // 是否已连接到气门嘴
        private bool _isPressurizing;              // 是否正在加压
        private Coroutine _pressurizeCoroutine;    // 加压协程引用
        
        #endregion
        
        #region 公共属性 - 供外部访问
        
        /// <summary>
        /// 当前胎压值（bar）
        /// </summary>
        public float CurrentPressure => _currentPressure;
        
        /// <summary>
        /// 是否已连接到气门嘴
        /// </summary>
        public bool IsConnected => _isConnected;
        
        /// <summary>
        /// 胎压是否正常（达到标准值）
        /// </summary>
        public bool IsPressureNormal => _currentPressure >= _standardPressure;
        
        #endregion
        
        #region 事件 - 供桌面小组复用
        
        /// <summary>
        /// 胎压值变化事件（参数：当前胎压）
        /// </summary>
        public System.Action<float> OnPressureChanged;
        
        /// <summary>
        /// 连接状态变化事件（参数：是否已连接）
        /// </summary>
        public System.Action<bool> OnConnectionChanged;
        
        /// <summary>
        /// 胎压达到标准值事件
        /// </summary>
        public System.Action OnPressureNormalReached;
        
        /// <summary>
        /// 胎压超过最大值事件
        /// </summary>
        public System.Action OnPressureExceeded;
        
        #endregion
        
        #region Unity生命周期
        
        private void Start()
        {
            Initialize();
            RegisterEvents();
        }
        
        private void OnDestroy()
        {
            UnregisterEvents();
            Cleanup();
        }
        
        #endregion
        
        #region 初始化与清理
        
        /// <summary>
        /// 初始化组件状态
        /// </summary>
        private void Initialize()
        {
            _currentPressure = _initialPressure;
            _isConnected = false;
            _isPressurizing = false;
            UpdateDisplay();
        }
        
        /// <summary>
        /// 注册交互事件
        /// </summary>
        private void RegisterEvents()
        {
            if (_pressurizeButton != null)
            {
                _pressurizeButton.selectEntered.AddListener(OnPressurizeButtonPressed);
                _pressurizeButton.selectExited.AddListener(OnPressurizeButtonReleased);
            }
            
            if (_connectTrigger != null)
            {
                // 触发器事件需要通过单独的脚本或OnTriggerEnter处理
                // 此处建议创建一个独立的触发器检测脚本
            }
        }
        
        /// <summary>
        /// 注销事件，防止内存泄漏
        /// </summary>
        private void UnregisterEvents()
        {
            if (_pressurizeButton != null)
            {
                _pressurizeButton.selectEntered.RemoveListener(OnPressurizeButtonPressed);
                _pressurizeButton.selectExited.RemoveListener(OnPressurizeButtonReleased);
            }
        }
        
        /// <summary>
        /// 清理协程等资源
        /// </summary>
        private void Cleanup()
        {
            if (_pressurizeCoroutine != null)
            {
                StopCoroutine(_pressurizeCoroutine);
                _pressurizeCoroutine = null;
            }
        }
        
        #endregion
        
        #region 公共方法 - 供外部调用（桌面小组重点适配）
        
        /// <summary>
        /// 连接到气门嘴时调用
        /// 【桌面小组适配】VR中通过触发器进入触发，桌面版可通过鼠标点击气门嘴触发
        /// </summary>
        public void ConnectToValve()
        {
            if (_isConnected) return;
            
            _isConnected = true;
            UpdateDisplay();
            OnConnectionChanged?.Invoke(true);
            
            // 发送全局事件，通知步骤管理器
            EventManager.TriggerEvent("TireGaugeConnected", this);
        }
        
        /// <summary>
        /// 断开气门嘴连接时调用
        /// </summary>
        public void DisconnectFromValve()
        {
            if (!_isConnected) return;
            
            // 断开时停止加压
            StopPressurize();
            _isConnected = false;
            OnConnectionChanged?.Invoke(false);
        }
        
        /// <summary>
        /// 开始加压（长按时调用）
        /// 【桌面小组适配】VR中由手柄按钮按下触发，桌面版可由鼠标按下触发
        /// </summary>
        public void StartPressurize()
        {
            if (!_isConnected)
            {
                Debug.LogWarning("未连接到气门嘴，无法加压");
                return;
            }
            
            if (_isPressurizing) return;
            
            _isPressurizing = true;
            if (_pressurizeCoroutine != null) StopCoroutine(_pressurizeCoroutine);
            _pressurizeCoroutine = StartCoroutine(PressurizeCoroutine());
        }
        
        /// <summary>
        /// 停止加压（松开时调用）
        /// </summary>
        public void StopPressurize()
        {
            if (!_isPressurizing) return;
            
            _isPressurizing = false;
            if (_pressurizeCoroutine != null)
            {
                StopCoroutine(_pressurizeCoroutine);
                _pressurizeCoroutine = null;
            }
        }
        
        #endregion
        
        #region 私有方法
        
        /// <summary>
        /// 加压协程
        /// </summary>
        private System.Collections.IEnumerator PressurizeCoroutine()
        {
            float stepTime = 0.05f;      // 每帧更新时间
            float stepValue = _pressurizeRate * stepTime;  // 每步增加量
            
            while (_currentPressure < _maxPressure)
            {
                _currentPressure += stepValue;
                _currentPressure = Mathf.Min(_currentPressure, _maxPressure);
                
                UpdateDisplay();
                PlayPressurizeSound();
                OnPressureChanged?.Invoke(_currentPressure);
                
                // 检查是否达到标准胎压
                if (_currentPressure >= _standardPressure && !IsPressureNormal)
                {
                    OnPressureNormalReached?.Invoke();
                    EventManager.TriggerEvent("TirePressureNormal");
                    PlayCompleteSound();
                }
                
                yield return new WaitForSeconds(stepTime);
            }
            
            // 超过最大胎压
            if (_currentPressure >= _maxPressure)
            {
                OnPressureExceeded?.Invoke();
                EventManager.TriggerEvent("TirePressureExceeded");
                PlayWarningSound();
                _isPressurizing = false;
            }
        }
        
        /// <summary>
        /// 更新显示数值
        /// </summary>
        private void UpdateDisplay()
        {
            if (_displayText != null)
            {
                string status = _isConnected ? "" : " (未连接)";
                _displayText.text = $"{_currentPressure:F1} bar{status}";
                
                // 根据胎压值改变文字颜色
                if (_currentPressure >= _maxPressure)
                    _displayText.color = Color.red;
                else if (_currentPressure >= _standardPressure)
                    _displayText.color = Color.green;
                else
                    _displayText.color = Color.white;
            }
        }
        
        /// <summary>
        /// 播放加压音效
        /// </summary>
        private void PlayPressurizeSound()
        {
            if (_pressurizeSound != null && AudioManager.Instance != null)
            {
                AudioManager.Instance.PlayOneShot(_pressurizeSound, 0.5f);
            }
        }
        
        /// <summary>
        /// 播放完成音效
        /// </summary>
        private void PlayCompleteSound()
        {
            if (_completeSound != null && AudioManager.Instance != null)
            {
                AudioManager.Instance.PlayOneShot(_completeSound, 0.8f);
            }
        }
        
        /// <summary>
        /// 播放警告音效
        /// </summary>
        private void PlayWarningSound()
        {
            if (_warningSound != null && AudioManager.Instance != null)
            {
                AudioManager.Instance.PlayOneShot(_warningSound, 1.0f);
            }
        }
        
        #endregion
        
        #region 交互事件回调（VR专用，桌面小组需替换）
        
        /// <summary>
        /// 加压按钮按下回调（VR射线触发）
        /// 【桌面小组适配】将此逻辑改为鼠标OnPointerDown事件
        /// </summary>
        private void OnPressurizeButtonPressed(SelectEnterEventArgs args)
        {
            StartPressurize();
        }
        
        /// <summary>
        /// 加压按钮释放回调（VR射线触发）
        /// 【桌面小组适配】将此逻辑改为鼠标OnPointerUp事件
        /// </summary>
        private void OnPressurizeButtonReleased(SelectExitEventArgs args)
        {
            StopPressurize();
        }
        
        /// <summary>
        /// 触发器进入事件（连接气门嘴）
        /// 需要在单独的脚本中调用ConnectToValve()方法
        /// </summary>
        private void OnTriggerEnter(Collider other)
        {
            if (other.CompareTag("TireValve"))
            {
                ConnectToValve();
            }
        }
        
        /// <summary>
        /// 触发器退出事件（断开连接）
        /// </summary>
        private void OnTriggerExit(Collider other)
        {
            if (other.CompareTag("TireValve"))
            {
                DisconnectFromValve();
            }
        }
        
        #endregion
    }
}
